def check(name,ok):

def sql(q):
 r=subprocess.run(['docker','exec',kb.postgres,'psql','-U','postgres','-d','aimee_store','-X','-qAt','-v','ON_ERROR_STOP=1','-c',q],capture_output=True,text=True)

  # These SQL statements contain only fixture literals and schema names.
  reason=next((line for line in r.stderr.splitlines() if line.startswith('ERROR:')),'SQL operation failed')
  (out/'sql-error.json').write_text(json.dumps(dict(reason=reason))+'\n')
  raise RuntimeError(reason)

def action(verb,body):
 started=time.monotonic();status,data=kb.kb_request('/v1/actions/memory.'+verb,body)
 calls.append(dict(verb=verb,http_status=status,seconds=time.monotonic()-started,response=data));(out/'responses.json').write_text(json.dumps(calls,indent=2)+'\n')
 if status!=200 or data.get('status') in ['error','unavailable']:raise RuntimeError('owner action failed: '+verb)
 return data
def settle():
 deadline=time.monotonic()+150
 while time.monotonic()<deadline:
  if sql("SELECT (SELECT count(*) FROM kb_async_jobs WHERE kind IN ('memory_index','memory_assertion_index') AND status<>'done')+(SELECT count(*) FROM vector_index_ops WHERE status IN ('pending','failed'))")=='0':return
  time.sleep(1)
 raise RuntimeError('canonical indexing did not settle')
def prepare(version):

  d=action('reembed_start',dict(version=version,embedding_command=env['EMBEDDER_URL'],scope='all'))
  if d.get('failed',0):raise RuntimeError('generation backfill failed')
  if d.get('ready'):
   settle();return d
 raise RuntimeError('generation backfill did not finish')
